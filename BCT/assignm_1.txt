pragma solidity ^0.8.20;

contract MyBank
{
    mapping(address => uint256) private _balances;
    address public owner;

    event LogDepositMade(address accountHolder, uint256 amount);
    event LogWithdrawalMade(address accountHolder, uint256 amount);

    constructor()
    {
        owner = msg.sender;
    }

    // Deposit ETH
    function deposit() public payable returns (uint256)
    {
        require(msg.value > 0, "Deposit amount must be greater than 0");

        _balances[msg.sender] += msg.value;

        emit LogDepositMade(msg.sender, msg.value);

        return _balances[msg.sender];
    }

    // Withdraw ETH
    function withdraw(uint256 withdrawAmount) public returns (uint256)
    {
        require(withdrawAmount > 0, "Amount must be greater than 0");
        require(
            _balances[msg.sender] >= withdrawAmount,
            "Insufficient balance"
        );

        _balances[msg.sender] -= withdrawAmount;

        // Transfer ETH back to the user's wallet
        (bool success, ) = payable(msg.sender).call{value: withdrawAmount}("");

        require(success, "ETH transfer failed");

        emit LogWithdrawalMade(msg.sender, withdrawAmount);

        return _balances[msg.sender];
    }

    // View balance
    function viewBalance() public view returns (uint256)
    {
        return _balances[msg.sender];
    }
}
